        const targetPositions = new Set(positions.slice(0, numTargets));

        for (let i = 0; i < 25; i++) {
            const cell = document.createElement('div');
            cell.className = 'grid-cell bg-white shadow';
            let letter;
            if (targetPositions.has(i)) {
                letter = target;
                targetCount++;
            } else {
                do { letter = allLetters[Math.floor(Math.random()*allLetters.length)].replace('هـ','ه'); } while (letter === target);
            }
            cell.textContent = letter;
            cell.dataset.isTarget = targetPositions.has(i) ? '1' : '0';
            cell.onclick = () => {
                if (cell.classList.contains('correct') || cell.classList.contains('wrong')) return;
                if (cell.dataset.isTarget === '1') {
                    cell.classList.add('correct');
                    playCorrect();
                    targetCount--;
                    if (targetCount <= 0) { addStar(); showConfetti(); markCompleted(); }
                } else {
                    cell.classList.add('wrong');
                    playWrong();
                    setTimeout(() => cell.classList.remove('wrong'), 500);
                }
            };
            grid.appendChild(cell);
        }
        // Fix targetCount to actual
        targetCount = numTargets;
        container.appendChild(grid);
    }

    // Activity 4: Drag letter to image
    function activity4(item, container) {
        const prompt = document.createElement('p');
        prompt.className = 'text-lg font-bold text-center';
        prompt.textContent = 'اسحب الحرف الصحيح إلى الصورة';
        container.appendChild(prompt);

        const imgDiv = document.createElement('div');
        imgDiv.className = 'drop-zone flex flex-col items-center justify-center p-4 mt-4';
        imgDiv.innerHTML = `<img src="${item.img}" class="w-32 h-32 object-cover rounded-xl" loading="lazy"><p class="mt-2 font-bold">${item.emoji} ${item.word}</p>`;
        imgDiv.ondragover = e => { e.preventDefault(); imgDiv.classList.add('over'); };
        imgDiv.ondragleave = () => imgDiv.classList.remove('over');
        imgDiv.ondrop = e => {
            e.preventDefault(); imgDiv.classList.remove('over');
            const letter = e.dataTransfer.getData('text/plain');
            if (letter === item.letter.replace('هـ','ه')) {
                playCorrect(); addStar(); showConfetti(); markCompleted();
                imgDiv.style.borderColor = '#22c55e'; imgDiv.style.background = '#dcfce7';
            } else { playWrong(); }
        };
        container.appendChild(imgDiv);

        const letters = [item.letter.replace('هـ','ه')];
        while (letters.length < 4) {
            const r = allLetters[Math.floor(Math.random()*allLetters.length)].replace('هـ','ه');
            if (!letters.includes(r)) letters.push(r);
        }
        letters.sort(() => Math.random()-0.5);

        const row = document.createElement('div');
        row.className = 'flex gap-3 mt-6 justify-center';
        letters.forEach(l => {
            const el = document.createElement('div');
            el.className = 'letter-circle bg-white shadow-lg draggable text-2xl';
            el.textContent = l;
            el.draggable = true;
            el.ondragstart = e => e.dataTransfer.setData('text/plain', l);
            // Touch support
            let touchClone = null;
            el.ontouchstart = e => {
                const touch = e.touches[0];
                touchClone = el.cloneNode(true);
                touchClone.style.position = 'fixed';
                touchClone.style.zIndex = '9999';
                touchClone.style.pointerEvents = 'none';
                touchClone.style.opacity = '0.8';
                document.body.appendChild(touchClone);
                moveTouchClone(touch);
            };
            el.ontouchmove = e => { e.preventDefault(); moveTouchClone(e.touches[0]); };
            el.ontouchend = e => {
                if (touchClone) { touchClone.remove(); touchClone = null; }
                const touch = e.changedTouches[0];
                const dropEl = document.elementFromPoint(touch.clientX, touch.clientY);
                if (dropEl && (dropEl === imgDiv || imgDiv.contains(dropEl))) {
                    if (l === item.letter.replace('هـ','ه')) {
                        playCorrect(); addStar(); showConfetti(); markCompleted();
                        imgDiv.style.borderColor = '#22c55e'; imgDiv.style.background = '#dcfce7';
                    } else { playWrong(); }
                }
            };
            function moveTouchClone(t) { if(touchClone){touchClone.style.left=(t.clientX-30)+'px';touchClone.style.top=(t.clientY-30)+'px';} }
            row.appendChild(el);
        });
        container.appendChild(row);
    }

    // Activity 5: Arrange word
    function activity5(item, container) {
        const word = item.word;
        const letters = word.split('').sort(() => Math.random()-0.5);

        const prompt = document.createElement('div');
        prompt.className = 'text-center';
        prompt.innerHTML = `<p class="text-4xl mb-2">${item.emoji}</p><p class="text-lg font-bold">رتب الحروف لتكوين الكلمة</p>`;
        container.appendChild(prompt);

        const answerSlots = document.createElement('div');
        answerSlots.className = 'flex gap-2 justify-center mt-4';
        const slots = [];
        for (let i = 0; i < word.length; i++) {
            const slot = document.createElement('div');
            slot.className = 'w-12 h-12 border-2 border-purple-300 rounded-xl flex items-center justify-center text-2xl font-bold bg-white';
            slot.dataset.index = i;
            answerSlots.appendChild(slot);
            slots.push(slot);
        }
        container.appendChild(answerSlots);

        let currentIndex = 0;
        const letterBtns = document.createElement('div');
        letterBtns.className = 'flex gap-2 justify-center flex-wrap mt-6';
        letters.forEach((l, i) => {
            const btn = document.createElement('button');
            btn.className = 'letter-circle bg-white shadow-md text-xl';
            btn.textContent = l;
            btn.onclick = () => {
                if (currentIndex >= word.length) return;
                slots[currentIndex].textContent = l;
                btn.classList.add('opacity-30');
                btn.disabled = true;
                currentIndex++;
                if (currentIndex === word.length) {
                    const result = slots.map(s => s.textContent).join('');
                    if (result === word) {
                        playCorrect(); addStar(); showConfetti(); markCompleted();
                    } else {
                        playWrong();
                        setTimeout(() => {
                            currentIndex = 0;
                            slots.forEach(s => s.textContent = '');
                            letterBtns.querySelectorAll('button').forEach(b => { b.classList.remove('opacity-30'); b.disabled = false; });
                        }, 800);
                    }
                }
            };
            letterBtns.appendChild(btn);
        });
        container.appendChild(letterBtns);
    }
